export const xs = Array.from({ length: 121 }, (_, i) => -6 + (12 * i) / 120);
export const gauss = (x, s) => Math.exp(-(x * x) / (2 * s * s)) / (s * Math.sqrt(2 * Math.PI));
export const round = (v) => Math.round(v * 1e5) / 1e5;
// Heat equation $\partial_t p = \partial_x^2 p$ started at N(0, 1/4): variance 1/4 + 2t.

  name: t.toFixed(1),
  data: [{ x: xs, y: xs.map((x) => round(gauss(x, sigmaAt(t)))) }],
}));

## A theorem, with math

:::proposition[Gaussian heat flow]{#prp-heat}
Let $p_0 = \mathcal N(0, \sigma_0^2)$ and let $p_t$ solve the heat equation $\partial_t p_t = \partial_x^2 p_t$. Then
$$
\begin{equation}
    p_t = \mathcal N(0, \sigma_0^2 + 2t).
\end{equation}
$$
:::

<details>
<summary>Proof</summary>

Take Fourier transforms in $x$. The equation becomes $\partial_t \hat p_t(\xi) = -\xi^2 \hat p_t(\xi)$, so
$\hat p_t(\xi) = e^{-\xi^2 t}\hat p_0(\xi)$. The Fourier transform of $\mathcal N(0, \sigma^2)$ is $e^{-\sigma^2\xi^2/2}$, hence the exponent
$-\tfrac12\sigma_0^2\xi^2 - t\xi^2 = -\tfrac12(\sigma_0^2 + 2t)\xi^2$ identifies the variance $\sigma_0^2 + 2t$.

</details>
